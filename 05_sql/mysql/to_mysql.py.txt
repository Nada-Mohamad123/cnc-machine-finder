import sqlite3
import pandas as pd
from sqlalchemy import create_engine

PASSWORD = "ضع_الباسورد_هنا"

engine = create_engine(
    f"mysql+pymysql://root:{PASSWORD}@localhost/machines_db?charset=utf8mb4"
)

from pathlib import Path
DB_FILE = Path(__file__).resolve().parents[2] / "04_database" / "machines.db"
with sqlite3.connect(DB_FILE) as conn:
    for table in ["machines", "specs_long"]:
        df = pd.read_sql_query(f"SELECT * FROM {table}", conn)
        df.to_sql(table, engine, if_exists="replace", index=False, chunksize=500)
        print(table, len(df), "rows copied")
